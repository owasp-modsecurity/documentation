{{- /* Overrides the theme's one-line content partial to color inline code that is a complete SecLang directive (a Sec* directive followed by arguments). Hugo has no render hook for code spans, so the rendered content is post-processed: each such span becomes a one-line version of the seclang code block, on the same dark background so the Monokai colors stay readable in both theme variants. */ -}} {{- $c := printf "%s" .Content -}} {{- range $m := findRE `Sec[A-Za-z0-9]+\s+[^<]+` $c | uniq -}} {{- $inner := strings.TrimSuffix "" (strings.TrimPrefix "" $m) -}} {{- $inner = replace $inner """ """ -}} {{- $h := partial "seclang-highlight.html" (printf "%s" $inner) -}} {{- $c = replace $c $m (printf `%s` $h) -}} {{- end -}} {{- $c | safeHTML -}}
